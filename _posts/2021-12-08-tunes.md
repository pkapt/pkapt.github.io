<script>
  (function () {
    var storageKey = "pkapt-tune-trainer-ratings-v1";
    var tunes = Array.prototype.slice
      .call(document.querySelectorAll("#tunes-list a"))
      .map(function (link) {
        return {
          name: link.textContent.trim(),
          href: link.href
        };
      });

    var pickButton = document.getElementById("tune-trainer-pick");
    var resetButton = document.getElementById("tune-trainer-reset");
    var card = document.getElementById("tune-trainer-card");
    var tuneLink = document.getElementById("tune-trainer-link");
    var status = document.getElementById("tune-trainer-status");
    var currentTune = null;

    function loadRatings() {
      try {
        return JSON.parse(localStorage.getItem(storageKey)) || {};
      } catch (error) {
        return {};
      }
    }

    function saveRatings(ratings) {
      localStorage.setItem(storageKey, JSON.stringify(ratings));
    }

    function getWeight(tune, ratings) {
      var rating = Number(ratings[tune.name]);
      if (rating === 1) return 1;
      if (rating === 2) return 3;
      if (rating === 3) return 6;
      return 8;
    }

    function pickWeightedTune() {
      var ratings = loadRatings();
      var weightedTunes = tunes.map(function (tune) {
        return {
          tune: tune,
          weight: getWeight(tune, ratings)
        };
      });
      var totalWeight = weightedTunes.reduce(function (sum, item) {
        return sum + item.weight;
      }, 0);
      var target = Math.random() * totalWeight;

      for (var i = 0; i < weightedTunes.length; i += 1) {
        target -= weightedTunes[i].weight;
        if (target <= 0) {
          return weightedTunes[i].tune;
        }
      }

      return tunes[tunes.length - 1];
    }

    function updateStatus() {
      var ratings = loadRatings();
      var counts = { 1: 0, 2: 0, 3: 0 };
      var ratedCount = Object.keys(ratings).filter(function (name) {
        return tunes.some(function (tune) {
          return tune.name === name;
        });
      }).length;

      Object.keys(ratings).forEach(function (name) {
        if (counts[ratings[name]] !== undefined) {
          counts[ratings[name]] += 1;
        }
      });

      status.textContent =
        ratedCount + " of " + tunes.length + " rated. " +
        "Good: " + counts[1] + ", OK: " + counts[2] + ", Needs work: " + counts[3] + ".";
    }

    function showTune(tune) {
      currentTune = tune;
      tuneLink.textContent = tune.name;
      tuneLink.href = tune.href;
      card.hidden = false;
    }

    pickButton.addEventListener("click", function () {
      showTune(pickWeightedTune());
    });

    card.addEventListener("click", function (event) {
      var rating = event.target.getAttribute("data-rating");
      if (!rating || !currentTune) return;

      var ratings = loadRatings();
      ratings[currentTune.name] = Number(rating);
      saveRatings(ratings);
      updateStatus();
      showTune(pickWeightedTune());
    });

    resetButton.addEventListener("click", function () {
      if (!window.confirm("Clear all saved tune ratings in this browser?")) return;
      localStorage.removeItem(storageKey);
      card.hidden = true;
      currentTune = null;
      updateStatus();
    });

    updateStatus();
  })();
</script>
